fmt = (x, d = 0) => x.toLocaleString("en-GB", {maximumFractionDigits: d, minimumFractionDigits: d})
// ---- a small, general einsum in JavaScript -------------------------------------------------
parseSpec = spec => {
  const s = spec.replace(/\s+/g, "");
  if (!/^[A-Za-z,]*(->[A-Za-z]*)?$/.test(s)) throw new Error("Use letters, commas and one '->', e.g. ij,jk->ik");
  let [lhs, rhs] = s.split("->"); const ins = lhs.split(",");
  if (ins.some(t => t === "") && ins.length > 1) throw new Error("An operand has no indices");
  const count = {}; for (const c of ins.join("")) count[c] = (count[c] || 0) + 1;
  if (rhs === undefined) rhs = Object.keys(count).filter(c => count[c] === 1).sort().join("");
  for (const c of rhs) if (!(c in count)) throw new Error(`Output index '${c}' does not appear in any input`);
  if (new Set(rhs).size !== rhs.length) throw new Error("An output index is repeated");
  const letters = [...new Set(ins.join("") + rhs)];
  return {ins, out: rhs, letters, summed: letters.filter(c => !rhs.includes(c))};
}
einsum = (spec, ...ts) => {
  const p = parseSpec(spec);
  if (ts.length !== p.ins.length) throw new Error(`Spec has ${p.ins.length} operands, got ${ts.length}`);
  const size = {};
  p.ins.forEach((idx, k) => {
    if (idx.length !== ts[k].shape.length) throw new Error(`Operand ${k + 1} has ${ts[k].shape.length} axes but '${idx}' names ${idx.length}`);
    [...idx].forEach((c, a) => { const n = ts[k].shape[a]; if (size[c] !== undefined && size[c] !== n) throw new Error(`Index '${c}' has size ${size[c]} in one place and ${n} in another`); size[c] = n; });
  });
  const L = p.letters, nL = L.length, dims = L.map(c => size[c]);
  const total = dims.reduce((a, b) => a * b, 1);
  if (total > 3e6) throw new Error(`That is ${fmt(total)} terms; keep it under 3 million here`);
  const strides = (shape) => { const s = new Array(shape.length); let m = 1; for (let i = shape.length - 1; i >= 0; i--) { s[i] = m; m *= shape[i]; } return s; };
  const opStride = p.ins.map((idx, k) => { const st = strides(ts[k].shape), v = new Array(nL).fill(0); [...idx].forEach((c, a) => v[L.indexOf(c)] += st[a]); return v; });
  const oshape = [...p.out].map(c => size[c]), ost = strides(oshape), outStride = new Array(nL).fill(0);
  [...p.out].forEach((c, a) => outStride[L.indexOf(c)] = ost[a]);
  const out = new Float64Array(oshape.reduce((a, b) => a * b, 1)), ctr = new Array(nL).fill(0);
  const off = p.ins.map(() => 0); let oo = 0;
  for (let t = 0; t < total; t++) {
    let v = 1; for (let k = 0; k < ts.length; k++) v *= ts[k].data[off[k]];
    out[oo] += v;
    for (let i = nL - 1; i >= 0; i--) {                       // odometer increment
      ctr[i]++; for (let k = 0; k < ts.length; k++) off[k] += opStride[k][i]; oo += outStride[i];
      if (ctr[i] < dims[i]) break;
      for (let k = 0; k < ts.length; k++) off[k] -= opStride[k][i] * dims[i]; oo -= outStride[i] * dims[i]; ctr[i] = 0;
    }
  }
  return {shape: oshape, data: out, size, terms: total, p};
}
rng = seed => { let s = seed >>> 0 || 1; return () => (s = (s * 1664525 + 1013904223) >>> 0) / 4294967296; }
tensor = (shape, r, ints = true) => { const n = shape.reduce((a, b) => a * b, 1), d = new Float64Array(n); for (let i = 0; i < n; i++) d[i] = ints ? Math.floor(r() * 7) - 3 : r() * 2 - 1; return {shape, data: d}; }

loopCode = (p, size, names) => {
  const lines = [], pad = d => "    ".repeat(d), O = p.out.length ? `${names.out}[${[...p.out].join(", ")}]` : names.out;
  lines.push(`${names.out} = zeros(${p.out.length ? "(" + [...p.out].map(c => size[c]).join(", ") + (p.out.length === 1 ? ",)" : ")") : "()"})`);
  let d = 0;
  for (const c of p.out) lines.push(`${pad(d++)}for ${c} in range(${size[c]}):`);
  for (const c of p.summed) lines.push(`${pad(d++)}for ${c} in range(${size[c]}):        # summed`);
  const prod = p.ins.map((idx, k) => `${names.ins[k]}[${[...idx].join(", ")}]`).join(" * ");
  lines.push(`${pad(d)}${O} += ${prod}`);
  return lines.join("\n");
}

Every deep-learning library has a function that does nearly all of linear algebra from a short string. In NumPy, PyTorch and JAX it is called einsum, and

C = einsum("ij,jk->ik", A, B)

is matrix multiplication. The string names the axes of each operand with letters and says which letters survive into the result. It is worth learning properly, because once it is familiar the shapes in a model like a transformer stop being something to track in comments and become part of the code. This explainer covers the rule, a catalogue of the operations it expresses, a playground that evaluates any einsum in your browser and shows the loops it stands for, multi-head attention written in einsum, why the order of a contraction matters, and what its gradient is. It is a companion to the transformer series, which writes the same operations as matrix products.

The rule

Write a letter for every axis of every operand. Then:

  1. A letter that appears in the output is a free index: the result has an axis for it.
  2. A letter that appears in the inputs but not in the output is summed: the result is added up over every value it takes.
  3. Each entry of the result is the sum, over all the summed letters, of the product of the corresponding entries of the inputs.

Formally, for a spec in₁,in₂,…->out,

\[ \text{out}[\text{free}] = \sum_{\text{summed}} \;\prod_{k} \text{in}_k[\text{its letters}]. \]

So "ij,jk->ik" is \(C_{ik} = \sum_j A_{ij} B_{jk}\): i and k are free, j is summed. Letters that share a name must have the same size, since they range over the same values. Nothing else is needed. In particular, a letter can be shared between inputs and still survive, in which case the inputs are paired along that axis rather than summed over it, which is how batching works: "bij,bjk->bik" multiplies each of the b matrix pairs separately.

If the -> and output are omitted, NumPy and PyTorch use an implicit rule: the output is every letter that appears exactly once, in alphabetical order. That makes "ij,jk" a matrix product but "ba" a transpose and "ab" not, which is a reason to always write the output.

Three things in that list go beyond physics’ summation convention. A letter repeated within one operand, as in ii, walks the diagonal. A letter can appear in three or more operands, as in the bilinear form. And a letter shared by two inputs can be kept, as in batching and the elementwise product, which the classical convention cannot express at all.

Playground

Choose an operation or type your own. Shapes are given per operand, axes separated by x, operands by commas. The inputs are filled with small random integers so the arithmetic can be checked by hand; the playground shows the free and summed indices, the loops the einsum stands for, and the result.

The loop nest is the whole meaning of an einsum: one loop per distinct letter, the free letters outermost because they index the result, the summed letters inside, and a single multiply–accumulate in the middle. The number of terms is the product of the sizes of all the letters, which is why the count jumps when you add an index. Libraries do not run this loop; they rearrange the axes and hand the work to optimised matrix-multiplication routines, but the result is defined by it.

Multi-head attention in four lines

Attention is where einsum earns its keep, because the head axis makes plain matrix notation awkward. In the row-vector convention of the transformer series, with sequence positions t (or q and k when queries and keys are distinguished), model width m, heads h and head width d, and weight tensors \(W_Q, W_K, W_V \in \mathbb R^{m \times h \times d}\) and \(W_O \in \mathbb R^{h \times d \times m}\):

Q = einsum("tm,mhd->thd", X, W_Q)            # project every position into every head
K = einsum("tm,mhd->thd", X, W_K)
V = einsum("tm,mhd->thd", X, W_V)
S = einsum("qhd,khd->hqk", Q, K) / sqrt(d)   # scores: one T×T matrix per head
A = softmax(S + causal_mask, axis=-1)
Y = einsum("hqk,khd,hdm->qm", A, V, W_O)     # mix values, then project heads back and sum them

Each line says exactly what is contracted. The score line sums over the head width d only, and keeps the head axis while moving it to the front. The last line sums over keys k, head width d and heads h in one statement; summing over h is precisely “concatenate the heads and multiply by \(W_O\)”, without the concatenation. The matrix version of the same computation needs a reshape from \(T \times hd\) to \(T \times h \times d\), two transposes to bring the head axis forward, and the reverse on the way out, and those are where shape bugs live. Add a batch letter b to every operand except the weights, or write ..., and nothing else changes.

The computation below builds a random sequence and weights in your browser, runs the einsum version through the small einsum function this page uses, runs the matrix version with explicit reshapes and slices, and compares them:

viewof att = Inputs.form({T: Inputs.range([2, 10], {value: 6, step: 1, label: "Sequence length T"}), h: Inputs.range([1, 4], {value: 2, step: 1, label: "Heads h"}), d: Inputs.range([1, 8], {value: 4, step: 1, label: "Head width d"})})
attCheck = {
  const {T, h, d} = att, m = h * d, r = rng(11), X = tensor([T, m], r, false);
  const WQ = tensor([m, h, d], r, false), WK = tensor([m, h, d], r, false), WV = tensor([m, h, d], r, false), WO = tensor([h, d, m], r, false);
  const softmaxRows = (S, H) => { const out = new Float64Array(S.data.length);
    for (let hh = 0; hh < H; hh++) for (let q = 0; q < T; q++) { let mx = -Infinity; const base = (hh * T + q) * T;
      for (let k = 0; k <= q; k++) mx = Math.max(mx, S.data[base + k]); let z = 0;
      for (let k = 0; k <= q; k++) { out[base + k] = Math.exp(S.data[base + k] - mx); z += out[base + k]; }
      for (let k = 0; k <= q; k++) out[base + k] /= z; }
    return {shape: S.shape, data: out}; };
  // einsum version
  const Q = einsum("tm,mhd->thd", X, WQ), K = einsum("tm,mhd->thd", X, WK), V = einsum("tm,mhd->thd", X, WV);
  const S = einsum("qhd,khd->hqk", Q, K); S.data.forEach((v, i) => S.data[i] = v / Math.sqrt(d));
  const A = softmaxRows(S, h), Y = einsum("hqk,khd,hdm->qm", A, V, WO);
  // matrix version: per head, Q_h = X W_Q[:, h, :], scores = Q_h K_hᵀ, out_h = P V_h, concat, then concat · W_O(reshaped)
  const mm = (a, ar, ac, b, bc) => { const o = new Float64Array(ar * bc); for (let i = 0; i < ar; i++) for (let k = 0; k < ac; k++) { const av = a[i * ac + k]; for (let j = 0; j < bc; j++) o[i * bc + j] += av * b[k * bc + j]; } return o; };
  const slice = (W, hh) => { const o = new Float64Array(m * d); for (let i = 0; i < m; i++) for (let j = 0; j < d; j++) o[i * d + j] = W.data[(i * h + hh) * d + j]; return o; };
  const concat = new Float64Array(T * m);
  for (let hh = 0; hh < h; hh++) {
    const Qh = mm(X.data, T, m, slice(WQ, hh), d), Kh = mm(X.data, T, m, slice(WK, hh), d), Vh = mm(X.data, T, m, slice(WV, hh), d);
    const KhT = new Float64Array(d * T); for (let i = 0; i < T; i++) for (let j = 0; j < d; j++) KhT[j * T + i] = Kh[i * d + j];
    const Sh = mm(Qh, T, d, KhT, T).map(v => v / Math.sqrt(d)), P = softmaxRows({shape: [1, T, T], data: Sh}, 1).data;
    const Oh = mm(P, T, T, Vh, d); for (let i = 0; i < T; i++) for (let j = 0; j < d; j++) concat[i * m + hh * d + j] = Oh[i * d + j];
  }
  const Ym = mm(concat, T, m, WO.data, m);   // W_O (h, d, m) read as an (h·d) × m matrix
  const err = d3.max(Ym, (v, i) => Math.abs(v - Y.data[i]));
  return htl.html`<div class="es-panel"><h4>Multi-head attention, T = ${T}, h = ${h}, d = ${d}, m = ${m}</h4><div class="es-kpis">
    <div><div class="k">Output shape</div><div class="v">(${Y.shape.join(", ")})</div></div>
    <div><div class="k">Largest difference</div><div class="v">${err.toExponential(1)}</div><div class="s">einsum against reshape-and-matmul</div></div>
    <div><div class="k">Score tensor</div><div class="v">(${S.shape.join(", ")})</div><div class="s">one T × T matrix per head</div></div></div></div>`;
}

The two agree to rounding error. Reading \(W_O\), a tensor of shape \((h, d, m)\), as a matrix of shape \((hd) \times m\) is exactly what makes the concatenation unnecessary: the row-major layout puts head 0’s \(d\) rows first, then head 1’s, the same order as the concatenated head outputs.

The order of a contraction

An einsum with three or more operands can be evaluated in one loop nest, but that is rarely the cheap way. Take a chain of three matrices, "ij,jk,kl->il". The single loop nest touches every combination of i, j, k, l: \(IJKL\) multiplications, twice. Contracting a pair first and then the third is the familiar matrix-chain problem: \((AB)C\) costs \(IJK + IKL\), and \(A(BC)\) costs \(JKL + IJL\).

The same choice has a famous instance inside attention. Drop the softmax, so that attention becomes \((QK^\top)V\), a plain three-way contraction "qd,kd,ke->qe" over sequence length \(T\) and head width \(d\). Contracting queries with keys first builds the \(T \times T\) score matrix, costing about \(2T^2 d\). Contracting keys with values first builds a \(d \times d\) summary, "kd,ke->de", and costs about \(2Td^2\): linear in sequence length. That reordering is the whole idea of linear attention (Angelos Katharopoulos and colleagues, 2020), and the softmax, which is not multilinear, is exactly what prevents it in a standard transformer.

viewof chain = Inputs.form({

For a chain, dynamic programming finds the best order in polynomial time. For a general network of contractions, finding the cheapest order is NP-hard, a result due to Chi-Chung Lam, P. Sadayappan and Rephael Wenger (1997), so libraries use heuristics. NumPy’s einsum does so when called with optimize=True, and Daniel Smith and Johnnie Gray’s opt_einsum package (2018), which PyTorch uses when it is installed, offers greedy, dynamic-programming and randomised searches. Large tensor-network simulations of quantum circuits live or die by this choice.

Gradients are einsums too

An einsum is linear in each operand separately, so its derivative with respect to one operand is the einsum with that operand removed. For a scalar loss \(\mathcal L\) and \(C\) = einsum("ij,jk->ik", A, B),

\[ \frac{\partial \mathcal L}{\partial A_{ij}} = \sum_k \frac{\partial \mathcal L}{\partial C_{ik}} \, B_{jk}, \qquad\text{that is}\qquad \bar A = \texttt{einsum("ik,jk->ij"}, \bar C, B\texttt{)}. \]

The rule in general: to get the gradient for one operand, swap it with the output. Its index string becomes the new output, and the output’s string, carrying the incoming gradient \(\bar C\), takes its place among the inputs. Summed letters become free and free letters become summed as needed, automatically.

i,j->ij ij,j->i i,ij->j

The rule needs two provisos. If an operand has a letter that appears nowhere else, such as j in "ij->i", the forward pass summed it away, and the gradient must broadcast it back, so the backward einsum has to be told that letter’s size. And a letter repeated within one operand, as in a trace, puts the gradient on a diagonal and zeros elsewhere. Automatic-differentiation systems handle both; the table covers the cases that matter in practice.

In practice

The same string everywhere. numpy.einsum, torch.einsum, jax.numpy.einsum and tf.einsum accept the same notation. All support ... for “any leading axes”, so "...ij,...jk->...ik" is a matrix product broadcast over any batch shape. Under the hood, a two-operand einsum is turned into a transpose, a reshape and a single batched matrix multiplication whenever possible, so it runs at the speed of the matrix-multiplication routine; an einsum is not slow because it is written as a string.

einops. Alex Rogozhnikov’s einops library (2018; ICLR 2022) took the idea further with named axes, grouping in parentheses, and separate verbs for rearranging, reducing and repeating. Splitting and merging heads becomes

Q = rearrange(Q, "b t (h d) -> b h t d", h=n_heads)   # split heads
Y = rearrange(Y, "b h t d -> b t (h d)")              # merge them back

and the reshape that the matrix version of attention needs is stated, checked against the sizes, and documented in one line. Its einsum accepts whole words for axes, "batch head query dim, batch head key dim -> batch head query key", at the cost of length.

What einsum cannot do. It only multiplies and adds, so every operation it expresses is multilinear. Softmax, normalisation, activation functions, maxima and anything else nonlinear sit between einsums. It also has no notion of a metric: physics’ distinction between upper and lower indices, which records how a quantity changes under a change of coordinates, has no counterpart, and the string says nothing about what the axes mean.

Where it comes from

Albert Einstein introduced the convention in 1916, in Die Grundlage der allgemeinen Relativitätstheorie, to tame equations of general relativity in which nearly every term was a sum over repeated coordinate indices: write the index twice, once up and once down, and drop the summation sign. It built on the absolute differential calculus of Gregorio Ricci-Curbastro and Tullio Levi-Civita (1900). The computing version keeps the idea of summing over repeated letters and drops the rest. NumPy added einsum in version 1.6, in 2011, and the deep-learning libraries followed. Tim Rocktäschel’s 2018 blog post Einsum is All You Need did much to make it everyday deep-learning vocabulary.
